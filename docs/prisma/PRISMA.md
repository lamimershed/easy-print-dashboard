## Overview

- **Module:** `src/prisma/prisma.module.ts`
- **Service:** `src/prisma/prisma.service.ts`
- **Database:** PostgreSQL (via `@prisma/adapter-pg` driver adapter)
- **Schema:** `prisma/schema.prisma`
- **Generated client output:** `src/generated/prisma/`
- **`PrismaModule` is global** — import once in `AppModule`, available everywhere without re-importing

---

## Architecture

```
PrismaService  (extends PrismaClient)
      │
      └── @prisma/adapter-pg  ──────► PostgreSQL (DATABASE_URL)
```

`PrismaService` implements `OnModuleInit` and `OnModuleDestroy` to manage the connection lifecycle. The `PrismaPg` adapter is used instead of the default query engine binary, which reduces cold-start time and is required for Railway's PostgreSQL integration.

## PrismaService

**File:** `src/prisma/prisma.service.ts`

```ts
@Injectable()
export class PrismaService extends PrismaClient implements OnModuleInit, OnModuleDestroy {
  async onModuleInit(): Promise<void> {
    await this.$connect();
  }
  async onModuleDestroy(): Promise<void> {
    await this.$disconnect();
  }
}
```

The constructor reads `DATABASE_URL` from `process.env` and passes it to the `PrismaPg` adapter.

**Usage in a service:**

```ts
constructor(private readonly prisma: PrismaService) {}

// Example query
const client = await this.prisma.client.findUnique({ where: { id } });
```

### Enums

#### `UserRole`

```prisma
enum UserRole {
  CLIENT        // Default role — print client owners
  SUPER_ADMIN   // Platform administrators
}
```

#### `ClientPlan`

```prisma
enum ClientPlan {
  FREE      // Default
  STARTER
  PRO
}
```

#### `SessionStatus`

```prisma
enum SessionStatus {
  WAITING     // client connected, no customer yet
  CONNECTED   // Customer has joined
  PRINTING    // File transfer complete, client printing
  COMPLETED   // Print confirmed by client
  EXPIRED     // Session TTL elapsed (set by cleanup job — not yet implemented)
}
```

#### `PrintJobStatus`

```prisma
enum PrintJobStatus {
  PENDING
  PRINTING    // Set when customer sends file metadata
  COMPLETED   // Set when client confirms print
  FAILED
}
```

#### `PaymentStatus`

```prisma
enum PaymentStatus {
  PENDING
  SUCCEEDED
  FAILED
  REFUNDED
}
```

### Models

#### `User`

Core account table. Linked one-to-one with either `Client` or `AdminUser`.

```prisma
model User {
  id           String   @id @default(cuid())
  email        String   @unique
  passwordHash String
  role         UserRole @default(CLIENT)
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  client        Client?
  admin         AdminUser?
  refreshTokens RefreshToken[]

  @@index([email])
}
```

#### `AdminUser`

Extends `User` for admin accounts. Created via `pnpm seed:admin` — no public registration.

```prisma
model AdminUser {
  id        String   @id @default(cuid())
  userId    String   @unique
  user      User     @relation(...)
  name      String
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}
```

#### `Client`

Print client profile. Created automatically during user registration.

```prisma
model Client {
  id          String     @id @default(cuid())
  userId      String     @unique
  slug        String     @unique   // URL-safe ID (auto-generated)
  companyName String
  phoneNumber String
  plan        ClientPlan @default(FREE)

  // Optional profile fields
  logoUrl           String?
  googleProfileLink String?
  address           String?
  latitude          Float?
  longitude         Float?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  sessions        Session[]
  printJobs       PrintJob[]
  payments        Payment[]
  analyticsEvents AnalyticsEvent[]

  @@index([userId])
  @@index([slug])
}
```

#### `RefreshToken`

Stores bcrypt hashes of active refresh tokens. Deleted on logout or rotation.

```prisma
model RefreshToken {
  id        String   @id @default(cuid())
  userId    String
  tokenHash String   @unique   // bcrypt hash of the raw token
  expiresAt DateTime
  createdAt DateTime @default(now())

  @@index([userId])
}
```

#### `Session`

Persisted record of a WebSocket print session. Written by `PrintService` when the first print job starts.

```prisma
model Session {
  id          String        @id @default(cuid())
  clientId    String?
  isEphemeral Boolean       @default(true)
  status      SessionStatus @default(WAITING)
  createdAt   DateTime      @default(now())
  expiresAt   DateTime
  completedAt DateTime?

  printJobs PrintJob[]

  @@index([clientId])
  @@index([status])
}
```

> The `Session.id` in the database is the same string as the in-memory session ID (8-char UUID prefix for ephemeral, or `clientId` for permanent). See `docs/session/SESSION.md`.

#### `PrintJob`

A single print request within a session.

```prisma
model PrintJob {
  id          String         @id @default(cuid())
  sessionId   String
  clientId    String?
  filename    String
  fileSize    Int
  mimeType    String
  status      PrintJobStatus @default(PENDING)
  createdAt   DateTime       @default(now())
  completedAt DateTime?

  @@index([clientId])
  @@index([sessionId])
  @@index([status])
}
```

#### `Payment`

Payment record linked to a client. Includes a `gatewayId` for payment provider references (e.g. Stripe PaymentIntent ID).

```prisma
model Payment {
  id          String        @id @default(cuid())
  clientId    String
  amountCents Int
  currency    String        @default("USD")
  status      PaymentStatus @default(PENDING)
  gatewayId   String?       @unique
  metadata    Json?
  createdAt   DateTime      @default(now())
  updatedAt   DateTime      @updatedAt

  @@index([clientId])
  @@index([status])
}
```

#### `AnalyticsEvent`

Append-only event log for client activity.

```prisma
model AnalyticsEvent {
  id        String   @id @default(cuid())
  clientId  String
  eventType String
  metadata  Json?
  createdAt DateTime @default(now())

  @@index([clientId, eventType])
  @@index([createdAt])
}
```

> ⚠️ This table has unbounded growth — see `docs/todo/DATABASE_SCALABILITY_TODO.md` for the retention policy plan.

## Generated Client

The Prisma client is generated into `src/generated/prisma/` (not the default `node_modules/.prisma/client`).

**Import paths:**

```ts
// Prisma Client (models, PrismaClient class)
import { PrismaClient, Client, User } from 'src/generated/prisma/client';

// Enums
import { UserRole, ClientPlan, SessionStatus } from 'src/generated/prisma/enums';
```

**Regenerate after schema changes:**

```bash
pnpm prisma generate
```

## Running Migrations

```bash
# Apply pending migrations (production)
pnpm prisma migrate deploy

# Create and apply a new migration (development)
pnpm prisma migrate dev --name describe-your-change

# Open Prisma Studio (DB browser)
pnpm prisma studio

# Re-generate the Prisma client after schema changes
pnpm prisma generate
```

The migration history lives in `prisma/migrations/`. Each migration is a timestamped folder with a `migration.sql` file.
